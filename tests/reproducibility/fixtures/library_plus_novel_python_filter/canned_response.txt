```novel_python
filter: |
  from dataclasses import dataclass
  from typing import Any
  from nanobrain.core.step import BaseStep, StepConfig


  @dataclass
  class ScoreThresholdFilterConfig(StepConfig):
      threshold: float = 0.5
      score_field: str = "score"


  class ScoreThresholdFilter(BaseStep):
      COMPONENT_TYPE = "score_threshold_filter"
      REQUIRED_CONFIG_FIELDS = ["name", "threshold", "score_field"]

      @classmethod
      def _get_config_class(cls):
          return ScoreThresholdFilterConfig

      async def process(self, inputs: dict[str, Any]) -> dict[str, Any]:
          rows = inputs["rows_input"]
          kept = [r for r in rows if float(r.get(self._score_field, 0.0)) >= self._threshold]
          return {"filtered_rows_output": kept}
```
